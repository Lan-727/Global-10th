<script>

const examples = {

  farm: {

    alternatives: [

      "Expand acreage",

      "Upgrade irrigation",

      "Keep current farm"

    ],

    states: [

      "High demand",

      "Stable demand",

      "Drought"

    ],

    payoffs: [

      [120,65,-45],

      [82,72,28],

      [48,48,48]

    ],

    probabilities: [0.35,0.40,0.25]

  },


  product: {

    alternatives: [

      "Premium launch",

      "Budget launch",

      "Delay launch",

      "License product"

    ],

    states: [

      "Strong market",

      "Average market",

      "Weak market"

    ],

    payoffs: [

      [180,55,-110],

      [95,62,8],

      [30,40,25],

      [105,75,15]

    ],

    probabilities: [0.25,0.50,0.25]

  },


  custom: {

    alternatives: ["Alternative 1","Alternative 2"],

    states: ["State 1","State 2"],

    payoffs: [[10,4],[6,8]],

    probabilities: [0.5,0.5]

  }

};


let data = structuredClone(examples.farm);


function escapeHTML(value) {

  return String(value).replace(/[&<>"']/g, c => ({

    "&":"&amp;","<":"&lt;",">":"&gt;",

    '"':"&quot;","'":"&#39;"

  }[c]));

}


function format(n) {

  return Number.isFinite(n)

    ? n.toLocaleString(undefined,{maximumFractionDigits:2})

    : "—";

}


function loadExample() {

  const key = document.getElementById("example").value;

  data = structuredClone(examples[key]);

  render();

}


function addAlternative() {

  data.alternatives.push(

    "Alternative " + (data.alternatives.length+1)

  );

  data.payoffs.push(data.states.map(()=>0));

  render();

}


function addState() {

  data.states.push("State " + (data.states.length+1));

  data.payoffs.forEach(row=>row.push(0));

  data.probabilities.push(0);

  render();

}


function removeAlternative(i) {

  if(data.alternatives.length<=1) {

    alert("At least one alternative is required.");

    return;

  }

  data.alternatives.splice(i,1);

  data.payoffs.splice(i,1);

  render();

}


function removeState(j) {

  if(data.states.length<=1) {

    alert("At least one state is required.");

    return;

  }

  data.states.splice(j,1);

  data.payoffs.forEach(row=>row.splice(j,1));

  data.probabilities.splice(j,1);

  render();

}


function valid() {

  const matrixOK =

    data.alternatives.length>0 &&

    data.states.length>0 &&

    data.payoffs.length===data.alternatives.length &&

    data.payoffs.every(row =>

      row.length===data.states.length &&

      row.every(Number.isFinite)

    );


  const probabilitySum =

    data.probabilities.reduce((a,b)=>a+b,0);


  const probabilitiesOK =

    data.probabilities.length===data.states.length &&

    data.probabilities.every(p=>Number.isFinite(p)&&p>=0) &&

    Math.abs(probabilitySum-1)<0.00000001;


  document.getElementById("validation").className =

    "notice "+(matrixOK?"success":"error");


  document.getElementById("validation").textContent =

    matrixOK ? "Payoff matrix is valid."

    : "Please enter a valid number in every payoff cell.";


  document.getElementById("probStatus").className =

    "notice "+(probabilitiesOK?"success":"error");


  document.getElementById("probStatus").textContent =

    "Probability total: "+

    format(probabilitySum*100)+"%"+

    (probabilitiesOK?" - Valid.":" - Must equal 100%.");


  return {matrixOK,probabilitiesOK};

}


function renderTables() {

  let html = `<thead><tr><th>Alternative / State</th>`;


  data.states.forEach((s,j)=>{

    html += `

      <th>

        <input class="name" value="${escapeHTML(s)}"

          onchange="renameState(${j},this.value)">

        <button class="remove" onclick="removeState(${j})">×</button>

      </th>`;

  });


  html += `<th>Action</th></tr></thead><tbody>`;


  data.alternatives.forEach((a,i)=>{

    html += `

      <tr>

        <td>

          <input class="name" value="${escapeHTML(a)}"

            onchange="renameAlternative(${i},this.value)">

        </td>`;


    data.payoffs[i].forEach((v,j)=>{

      html += `

        <td>

          <input class="payoff" type="number" step="any"

            value="${v}"

            oninput="updatePayoff(${i},${j},this.value)">

        </td>`;

    });


    html += `

      <td>

        <button class="remove"

          onclick="removeAlternative(${i})">Remove</button>

      </td>

      </tr>`;

  });


  document.getElementById("payoffTable").innerHTML =

    html+"</tbody>";


  let probabilities = `

    <thead><tr>

      <th>State</th><th>Probability (%)</th>

    </tr></thead><tbody>`;


  data.states.forEach((s,j)=>{

    probabilities += `

      <tr>

        <td>${escapeHTML(s)}</td>

        <td>

          <input class="payoff" type="number"

            min="0" max="100" step="any"

            value="${data.probabilities[j]*100}"

            oninput="updateProbability(${j},this.value)">

        </td>

      </tr>`;

  });


  document.getElementById("probTable").innerHTML =

    probabilities+"</tbody>";

}


function renameAlternative(i,value) {

  data.alternatives[i] = value.trim() || "Alternative "+(i+1);

  render();

}


function renameState(j,value) {

  data.states[j] = value.trim() || "State "+(j+1);

  render();

}


function updatePayoff(i,j,value) {

  data.payoffs[i][j] =

    value.trim()==="" ? NaN : Number(value);

  calculate();

}


function updateProbability(j,value) {

  data.probabilities[j] =

    value.trim()==="" ? NaN : Number(value)/100;

  calculate();

}


function findWinners(values,mode) {

  const best = mode==="min"

    ? Math.min(...values)

    : Math.max(...values);


  return {

    best,

    indexes:values.map((v,i)=>

      Math.abs(v-best)<1e-9?i:-1

    ).filter(i=>i>=0)

  };

}


function calculate() {

  const status=valid();


  if(!status.matrixOK) {

    document.getElementById("results").innerHTML="";

    document.getElementById("stages").innerHTML="";

    document.getElementById("flow").innerHTML="";

    return;

  }


  const rows=data.payoffs;


  const maxima=rows.map(r=>Math.max(...r));

  const minima=rows.map(r=>Math.min(...r));


  const stateBest=data.states.map((s,j)=>

    Math.max(...rows.map(r=>r[j]))

  );


  const regrets=rows.map(r=>

    r.map((v,j)=>stateBest[j]-v)

  );


  const maximumRegrets=regrets.map(r=>Math.max(...r));


  const emv=rows.map(r=>

    status.probabilitiesOK

      ? r.reduce((sum,v,j)=>

          sum+v*data.probabilities[j],0)

      : NaN

  );


  const maximax=findWinners(maxima,"max");

  const maximin=findWinners(minima,"max");

  const minimax=findWinners(maximumRegrets,"min");


  const bayes=status.probabilitiesOK

    ? findWinners(emv,"max")

    : {best:NaN,indexes:[]};


  const criteria=[

    ["Maximax",maximax,"Highest possible payoff"],

    ["Maximin",maximin,"Best worst-case payoff"],

    ["Minimax Regret",minimax,"Lowest maximum regret"],

    ["EMV / Bayes",bayes,"Expected monetary value"]

  ];


  document.getElementById("results").innerHTML =

    criteria.map(([name,result,description])=>`

      <div class="metric">

        <h3>${name}</h3>

        <strong>${format(result.best)}</strong>

        <div class="winner">

          ${result.indexes.length

            ? result.indexes.map(i=>

                escapeHTML(data.alternatives[i])

              ).join(", ")

            : "No recommendation"}

        </div>

        <small>${description}</small>

      </div>

    `).join("");


  stagedFiltering(maxima,minima,maximumRegrets,emv,

    status.probabilitiesOK);

}


function stagedFiltering(maxima,minima,regrets,emv,probOK) {

  const criteria=[

    {

      name:"Stage 1 - Maximax",

      values:maxima,

      mode:"max",

      description:"Keep the highest possible payoff."

    },

    {

      name:"Stage 2 - Maximin",

      values:minima,

      mode:"max",

      description:"Keep the best worst-case payoff."

    },

    {

      name:"Stage 3 - Minimax Regret",

      values:regrets,

      mode:"min",

      description:"Keep the lowest maximum regret."

    },

    {

      name:"Stage 4 - EMV / Bayes",

      values:emv,

      mode:"max",

      description:"Keep the highest expected payoff."

    }

  ];


  let survivors=data.alternatives.map((_,i)=>i);

  let stages=[];


  criteria.forEach((criterion,index)=>{

    const before=[...survivors];


    if(index===3&&!probOK) {

      stages.push({

        ...criterion,before,after:[],best:NaN,valid:false

      });

      survivors=[];

      return;

    }


    const values=before.map(i=>criterion.values[i]);


    const best=criterion.mode==="min"

      ? Math.min(...values)

      : Math.max(...values);


    survivors=before.filter(i=>

      Math.abs(criterion.values[i]-best)<1e-9

    );


    stages.push({

      ...criterion,before,after:[...survivors],

      best,valid:true

    });

  });


  document.getElementById("stages").innerHTML =

    stages.map(stage=>`

      <div class="stage">

        <strong>${stage.name}</strong>

        <p>${stage.description}</p>

        <p>Criterion value: ${format(stage.best)}</p>

        ${stage.before.map(i=>`

          <span class="chip ${stage.after.includes(i)?"":"out"}">

            ${escapeHTML(data.alternatives[i])}

            ${stage.after.includes(i)?"- Retained":"- Eliminated"}

          </span>

        `).join("")}

        ${!stage.valid

          ? "<p>Valid probabilities totaling 100% are required.</p>"

          : ""}

      </div>

    `).join("");


  drawFlow(stages);

}


function drawFlow(stages) {

  let final=stages[3].valid

    ? stages[3].after

    : stages[2].after;


  let html=`

    <div class="node">

      <strong>START</strong><br>

      ${data.alternatives.length} Alternatives

      <br>${data.states.length} States of Nature

    </div>

    <div class="arrow">↓</div>

  `;


  stages.forEach(stage=>{

    html+=`

      <div class="node">

        <strong>${stage.name}</strong><br>

        ${stage.valid

          ? "Best value: "+format(stage.best)+

            "<br>"+stage.after.length+" alternatives retained"

          : "Waiting for valid probabilities"}

      </div>

      <div class="arrow">↓</div>

    `;

  });


  html+=`

    <div class="node final">

      <strong>FINAL OUTCOME</strong><br>

      ${final.length

        ? final.map(i=>escapeHTML(data.alternatives[i])).join(", ")

        : "Check probabilities to obtain a final result."}

    </div>

  `;


  document.getElementById("flow").innerHTML=html;

}


function render() {

  renderTables();

  calculate();

}


render();
